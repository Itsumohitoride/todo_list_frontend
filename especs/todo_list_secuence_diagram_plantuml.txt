@startuml
actor User as user
participant "UserService" as userService
participant "ListService" as listService
participant "TaskService" as taskService

user -> userService: createAccount(firstName, lastName, email, nickname, password)
userService --> user: accountCreatedConfirmation

user -> userService: login(email, password)
userService --> user: loginSuccess

user -> listService: createList(listName, color, type)
listService --> user: listCreatedConfirmation

user -> listService: changeListColor(listId, newColor)
listService --> user: colorChangedConfirmation

user -> taskService: createTask(listId, description, type)
taskService --> user: taskCreatedConfirmation

user -> taskService: markTaskAsComplete(taskId)
taskService --> user: taskCompletionConfirmation

user -> listService: getProgress(listId)
listService --> user: progressData

user -> listService: getStatistics()
listService --> user: statisticsData

user -> userService: logout()
userService --> user: logoutSuccess
@enduml
